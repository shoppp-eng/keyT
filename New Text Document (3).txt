local function findMap()
    for _, map in ipairs(MAPS) do
        for _, id in ipairs(map.placeIds) do
            if id == game.PlaceId or id == game.GameId then return map end
        end
    end
    local lowerName = gameName:lower()
    for _, map in ipairs(MAPS) do
        for _, word in ipairs(map.match) do
            if lowerName:find(word:lower(), 1, true) then return map end
        end
    end
    return nil
end

local function new(class, props, parent)
    local o = Instance.new(class)
    for k, v in pairs(props) do o[k] = v end
    o.Parent = parent
    return o
end

local function round(o, r)
    new("UICorner", {CornerRadius = r}, o)
end

local function stroke(o, color, thick)
    return new("UIStroke", {
        Color = color, Thickness = thick,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
    }, o)
end

local function box(parent, x, y, w, h, color, r)
    local f = new("Frame", {
        Position = UDim2.new(0, x, 0, y),
        Size = UDim2.new(0, w, 0, h),
        BackgroundColor3 = color,
        BorderSizePixel = 0,
    }, parent)
    if r then round(f, UDim.new(0, r)) end
    return f
end

local function text(parent, str, x, y, w, h, size, color, font, align)
    return new("TextLabel", {
        Position = UDim2.new(0, x, 0, y),
        Size = UDim2.new(0, w, 0, h),
        BackgroundTransparency = 1,
        Text = str,
        TextColor3 = color,
        Font = font or Enum.Font.Gotham,
        TextSize = size,
        TextXAlignment = align or Enum.TextXAlignment.Left,
    }, parent)
end

local function tween(obj, t, props)
    local tw = TweenService:Create(obj, TweenInfo.new(t, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), props)
    tw:Play()
    return tw
end

local function hover(btn, normal, over)
    btn.MouseEnter:Connect(function() tween(btn, 0.15, {BackgroundColor3 = over}) end)
    btn.MouseLeave:Connect(function() tween(btn, 0.15, {BackgroundColor3 = normal}) end)
end

-- พยายามเปิดลิงก์ในเบราว์เซอร์ (ขึ้นกับ executor ว่ารองรับไหม)
local function openLink(link)
    -- 1) ฟังก์ชันเปิดลิงก์ที่ executor บางตัวมีให้
    local env = (getgenv and getgenv()) or _G
    for _, name in ipairs({ "openurl", "open_url", "openbrowser", "open_browser" }) do
        local fn = env[name]
        if type(fn) == "function" and pcall(fn, link) then
            return true
        end
    end

    -- 2) ลิงก์เชิญ Discord: สั่งแอป Discord บนคอมให้เปิด (ใช้ได้เฉพาะ PC ที่เปิด Discord ค้างไว้)
    local code = link:match("discord%.gg/(%w+)")
    local req = request or http_request or (syn and syn.request)
    if code and req then
        local HttpService = game:GetService("HttpService")
        local ok, res = pcall(function()
            return req({
                Url = "http://127.0.0.1:6463/rpc?v=1",
                Method = "POST",
                Headers = {
                    ["Content-Type"] = "application/json",
                    ["Origin"] = "https://discord.com",
                },
                Body = HttpService:JSONEncode({
                    cmd = "INVITE_BROWSER",
                    args = { code = code },
                    nonce = HttpService:GenerateGUID(false),
                }),
            })
        end)
        if ok and type(res) == "table" and (res.Success or res.StatusCode == 200) then
            return true
        end
    end

    return false
end

local popupToken = 0

-- ปิดทันที ไม่พึ่ง tween (กันค้าง)
local function hidePopup()
    popupToken += 1
    popupOverlay.Visible = false
    popupOverlay.BackgroundTransparency = 1
    popupScale.Scale = 0.85
end

-- ผูกการกดหลายแบบ (Activated / Click / InputBegan) กันบาง executor ที่บางอีเวนต์ไม่ทำงาน
do
    local last = 0
    local function fire()
        if os.clock() - last < 0.25 then return end
        last = os.clock()
        hidePopup()
    end
    popupClose.Activated:Connect(fire)
    popupClose.MouseButton1Click:Connect(fire)
    popupClose.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            fire()
        end
    end)
end

local function showPopup(kind, msg)
    popupToken += 1
    local good = (kind == "ok")
    local col = good and C.green or C.yellow
    popupStroke.Color = col
    popupIconBg.BackgroundColor3 = good and C.greenBg or C.yellowBg
    popupIcon.TextColor3 = col
    popupIcon.Text = good and "✓" or "!"
    popupMsg.Text = msg
    popupOverlay.Visible = true
    popupOverlay.BackgroundTransparency = 1
    popupScale.Scale = 0.85
    tween(popupOverlay, 0.2, { BackgroundTransparency = 0.5 })
    tween(popupScale, 0.2, { Scale = 1 })
end

local function copyLink(link)
    -- คัดลอกลิงก์ไว้เสมอ เผื่อเปิดเองไม่ได้
    local copied = false
    local fn = setclipboard or toclipboard
    if fn then
        copied = pcall(fn, link)
    end

    if openLink(link) then
        showPopup("ok", "เปิดลิงก์แล้ว")
    elseif copied then
        showPopup("ok", "คัดลอกลิงก์แล้ว\nนำไปวางในเบราว์เซอร์")
    else
        showPopup("warn", "คัดลอกอัตโนมัติไม่ได้\nลิงก์: " .. link)
    end
end

-- ===== ลากย้าย (ลากที่ส่วนหัว) =====
do
    local dragging, dragStart, startPos
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            if input.Position.Y - frame.AbsolutePosition.Y < 100 then
                dragging = true
                dragStart = input.Position
                startPos = holder.Position
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                    end
                end)
            end
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            local d = input.Position - dragStart
            holder.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X,
                                        startPos.Y.Scale, startPos.Y.Offset + d.Y)
        end
    end)
end

    return s:match("^%x%x%x%x%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%-%x%x%x%x%x%x%x%x%x%x%x%x$") ~= nil
end

local function submitKey()
    if busy then return end

    local key = keyBox.Text:gsub("%s+", "")
    keyBox.Text = key

    if key == "" then
        showStatus("bad", "กรุณาใส่คีย์ก่อน")
        return
    end

    if not isUUID(key) then
        showStatus("bad", "รูปแบบคีย์ไม่ถูกต้อง")
        return
    end

    -- เลือกสคริปต์ตามแมพที่เข้าอยู่
    local map = findMap()
    if not map then
        showStatus("warn", "ยังไม่รองรับแมพนี้ PlaceId: " .. tostring(game.PlaceId))
        return
    end

    busy = true
    submit.Text = "กำลังโหลด..."
    showStatus("warn", "กำลังโหลดสคริปต์ " .. map.name .. "...")

    getgenv().SCRIPT_KEY = key

    task.spawn(function()
        local ok, err = pcall(function()
            loadstring(game:HttpGet(map.url))()
        end)
        if not ok then warn("Script error:", err) end
    end)

    task.wait(1.5)
    showStatus("ok", "ส่งคีย์เรียบร้อย")
    task.wait(0.5)
    gui:Destroy()
end

submit.Activated:Connect(submitKey)
keyBox.FocusLost:Connect(function(enterPressed)
    tween(inputStroke, 0.2, {Color = C.line})
    if enterPressed then submitKey() end
end)
